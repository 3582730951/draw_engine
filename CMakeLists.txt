cmake_minimum_required(VERSION 3.16)

project(midraw C CXX)

option(MIDRAW_BUILD_SHARED "Build shared library" ON)
option(MIDRAW_BUILD_STATIC "Build static library" ON)
option(MIDRAW_BUILD_BENCHMARK "Build benchmark executable" ON)
option(MIDRAW_BUILD_OVERLAY "Build overlay engine executable" ON)
option(MIDRAW_BUILD_DEMO "Build draw demo executable" ON)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(CMAKE_BUILD_TYPE STREQUAL "Release")
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} -O3 -fomit-frame-pointer -ffunction-sections -fdata-sections")
  set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O3 -fomit-frame-pointer -ffunction-sections -fdata-sections")
  set(CMAKE_EXE_LINKER_FLAGS_RELEASE "${CMAKE_EXE_LINKER_FLAGS_RELEASE} -Wl,--gc-sections")
  set(CMAKE_SHARED_LINKER_FLAGS_RELEASE "${CMAKE_SHARED_LINKER_FLAGS_RELEASE} -Wl,--gc-sections")
endif()

set(MIDRAW_SOURCES
    ${CMAKE_CURRENT_LIST_DIR}/src/engine/midraw.cpp
    ${CMAKE_CURRENT_LIST_DIR}/src/engine/midraw_window.cpp
    ${CMAKE_CURRENT_LIST_DIR}/src/engine/draw_engine.cpp
)

set(MIDRAW_PUBLIC_HEADERS
    ${CMAKE_CURRENT_LIST_DIR}/include/midraw.h
    ${CMAKE_CURRENT_LIST_DIR}/include/draw_engine.h
)

set(FONT8X8_ROOT "" CACHE PATH "Path to font8x8 install prefix")
set(STB_TRUETYPE_ROOT "" CACHE PATH "Path to stb_truetype install prefix")

find_path(FONT8X8_INCLUDE_DIR font8x8_basic.h
          HINTS ${FONT8X8_ROOT} ${CMAKE_PREFIX_PATH}
          PATH_SUFFIXES include)
find_library(FONT8X8_LIBRARY font8x8
             HINTS ${FONT8X8_ROOT} ${CMAKE_PREFIX_PATH}
             PATH_SUFFIXES lib)

find_path(STB_TRUETYPE_INCLUDE_DIR stb_truetype.h
          HINTS ${STB_TRUETYPE_ROOT} ${CMAKE_PREFIX_PATH}
          PATH_SUFFIXES include)
find_library(STB_TRUETYPE_LIBRARY stb_truetype
             HINTS ${STB_TRUETYPE_ROOT} ${CMAKE_PREFIX_PATH}
             PATH_SUFFIXES lib)

find_library(VULKAN_LIB vulkan)
find_library(EGL_LIB EGL)
find_library(GLESV3_LIB GLESv3)
find_library(GLESV2_LIB GLESv2)
find_library(LOG_LIB log)

if(MIDRAW_BUILD_SHARED)
  add_library(midraw_shared SHARED ${MIDRAW_SOURCES})
  set_target_properties(midraw_shared PROPERTIES OUTPUT_NAME midraw)
  target_include_directories(midraw_shared PUBLIC ${CMAKE_CURRENT_LIST_DIR}/include)
  target_include_directories(midraw_shared PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src)
  target_include_directories(midraw_shared PRIVATE ${CMAKE_CURRENT_LIST_DIR}/generated)
  target_include_directories(midraw_shared PRIVATE ${CMAKE_CURRENT_LIST_DIR}/shaders)
  target_include_directories(midraw_shared PRIVATE ${CMAKE_CURRENT_LIST_DIR}/third_party)
  target_include_directories(midraw_shared PRIVATE ${CMAKE_CURRENT_LIST_DIR}/third_party/stb_truetype)
  target_link_libraries(midraw_shared PUBLIC dl)
  if(LOG_LIB)
    target_link_libraries(midraw_shared PUBLIC ${LOG_LIB})
  endif()
  if(VULKAN_LIB)
    target_link_libraries(midraw_shared PUBLIC ${VULKAN_LIB})
    target_compile_definitions(midraw_shared PRIVATE DRAW_ENGINE_HAS_VULKAN=1)
  else()
    target_compile_definitions(midraw_shared PRIVATE DRAW_ENGINE_HAS_VULKAN=0)
  endif()
  if(EGL_LIB AND (GLESV3_LIB OR GLESV2_LIB))
    target_link_libraries(midraw_shared PUBLIC ${EGL_LIB})
    if(GLESV3_LIB)
      target_link_libraries(midraw_shared PUBLIC ${GLESV3_LIB})
    else()
      target_link_libraries(midraw_shared PUBLIC ${GLESV2_LIB})
    endif()
    target_compile_definitions(midraw_shared PRIVATE DRAW_ENGINE_HAS_GLES=1)
  else()
    target_compile_definitions(midraw_shared PRIVATE DRAW_ENGINE_HAS_GLES=0)
  endif()
  if(FONT8X8_INCLUDE_DIR AND FONT8X8_LIBRARY)
    target_include_directories(midraw_shared PRIVATE ${FONT8X8_INCLUDE_DIR})
    target_link_libraries(midraw_shared PUBLIC ${FONT8X8_LIBRARY})
    target_compile_definitions(midraw_shared PRIVATE MIDRAW_USE_FONT8X8=1)
  endif()
  if(STB_TRUETYPE_INCLUDE_DIR AND STB_TRUETYPE_LIBRARY)
    target_include_directories(midraw_shared PRIVATE ${STB_TRUETYPE_INCLUDE_DIR})
    target_link_libraries(midraw_shared PUBLIC ${STB_TRUETYPE_LIBRARY})
    target_compile_definitions(midraw_shared PRIVATE MIDRAW_USE_STB_TRUETYPE=1)
  endif()
endif()

if(MIDRAW_BUILD_STATIC)
  add_library(midraw_static STATIC ${MIDRAW_SOURCES})
  set_target_properties(midraw_static PROPERTIES OUTPUT_NAME midraw)
  target_include_directories(midraw_static PUBLIC ${CMAKE_CURRENT_LIST_DIR}/include)

  target_include_directories(midraw_static PRIVATE ${CMAKE_CURRENT_LIST_DIR}/third_party)
  target_include_directories(midraw_static PRIVATE ${CMAKE_CURRENT_LIST_DIR}/third_party/stb_truetype)
  target_link_libraries(midraw_static PUBLIC dl)
  if(LOG_LIB)
    target_link_libraries(midraw_static PUBLIC ${LOG_LIB})
  endif()
  if(VULKAN_LIB)
    target_link_libraries(midraw_static PUBLIC ${VULKAN_LIB})
    target_compile_definitions(midraw_static PRIVATE DRAW_ENGINE_HAS_VULKAN=1)
  else()
    target_compile_definitions(midraw_static PRIVATE DRAW_ENGINE_HAS_VULKAN=0)
  endif()
  if(EGL_LIB AND (GLESV3_LIB OR GLESV2_LIB))
    target_link_libraries(midraw_static PUBLIC ${EGL_LIB})
    if(GLESV3_LIB)
      target_link_libraries(midraw_static PUBLIC ${GLESV3_LIB})
    else()
      target_link_libraries(midraw_static PUBLIC ${GLESV2_LIB})
    endif()
    target_compile_definitions(midraw_static PRIVATE DRAW_ENGINE_HAS_GLES=1)
  else()
    target_compile_definitions(midraw_static PRIVATE DRAW_ENGINE_HAS_GLES=0)
  endif()
  if(FONT8X8_INCLUDE_DIR AND FONT8X8_LIBRARY)
    target_include_directories(midraw_static PRIVATE ${FONT8X8_INCLUDE_DIR})
    target_link_libraries(midraw_static PUBLIC ${FONT8X8_LIBRARY})
    target_compile_definitions(midraw_static PRIVATE MIDRAW_USE_FONT8X8=1)
  endif()
  if(STB_TRUETYPE_INCLUDE_DIR AND STB_TRUETYPE_LIBRARY)
    target_include_directories(midraw_static PRIVATE ${STB_TRUETYPE_INCLUDE_DIR})
    target_link_libraries(midraw_static PUBLIC ${STB_TRUETYPE_LIBRARY})
    target_compile_definitions(midraw_static PRIVATE MIDRAW_USE_STB_TRUETYPE=1)
  endif()
endif()

if(MIDRAW_BUILD_BENCHMARK)
  add_executable(benchmark ${CMAKE_CURRENT_LIST_DIR}/apps/benchmark/main.cpp)
  target_include_directories(benchmark PRIVATE ${CMAKE_CURRENT_LIST_DIR}/include)
  if(TARGET midraw_static)
    target_link_libraries(benchmark PRIVATE midraw_static dl)
  elseif(TARGET midraw_shared)
    target_link_libraries(benchmark PRIVATE midraw_shared dl)
  endif()
endif()

if(MIDRAW_BUILD_DEMO)
  add_executable(draw_demo ${CMAKE_CURRENT_LIST_DIR}/apps/demo/main.cpp)
  target_include_directories(draw_demo PRIVATE ${CMAKE_CURRENT_LIST_DIR}/include)
  if(TARGET midraw_static)
    target_link_libraries(draw_demo PRIVATE midraw_static dl)
  elseif(TARGET midraw_shared)
    target_link_libraries(draw_demo PRIVATE midraw_shared dl)
  endif()
endif()

if(MIDRAW_BUILD_OVERLAY)
  add_executable(overlay_engine ${CMAKE_CURRENT_LIST_DIR}/apps/overlay/overlay_engine.cpp)
  target_include_directories(overlay_engine PRIVATE ${CMAKE_CURRENT_LIST_DIR}/include)
  target_link_libraries(overlay_engine PRIVATE dl)
endif()

install(FILES ${MIDRAW_PUBLIC_HEADERS} DESTINATION include)
if(TARGET midraw_shared)
  install(TARGETS midraw_shared DESTINATION lib)
endif()
if(TARGET midraw_static)
  install(TARGETS midraw_static DESTINATION lib)
endif()
